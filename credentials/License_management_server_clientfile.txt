using System;
using System.IO;
using System.Linq;
using System.Management;
using System.Net.NetworkInformation;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Collections.Generic;

public class MachineInfo
{
    public string MacAddress { get; set; } = "";
    public string SystemUuid { get; set; } = "";
    public string MotherboardSerial { get; set; } = "";
    public string BiosSerial { get; set; } = "";
    public string DiskSerial { get; set; } = "";
    public string Fingerprint { get; set; } = "";
}

public class LicenseData
{
    public string LicenseId { get; set; } = "";
    public string Fingerprint { get; set; } = "";
    public string Product { get; set; } = "";
    public string Expiry { get; set; } = "";
    public string Signature { get; set; } = "";
}

public static class ClientLicense
{
    /*
     * IMPORTANT:
     * Replace this with YOUR PUBLIC KEY generated by LicenseGenerator.cs.
     *
     * NEVER put your RSA PRIVATE KEY here.
     */
    private const string PUBLIC_KEY = @"
-----BEGIN PUBLIC KEY-----
PASTE_YOUR_PUBLIC_KEY_HERE
-----END PUBLIC KEY-----
";

    // ============================================================
    // MAIN FUNCTION
    // ============================================================

    public static MachineInfo GetMachineInfo()
    {
        var info = new MachineInfo();

        info.MacAddress = GetMacAddress();
        info.SystemUuid = GetSystemUuid();
        info.MotherboardSerial = GetMotherboardSerial();
        info.BiosSerial = GetBiosSerial();
        info.DiskSerial = GetDiskSerial();

        info.Fingerprint = GenerateFingerprint(info);

        return info;
    }

    // ============================================================
    // MAC ADDRESS
    // ============================================================

    private static string GetMacAddress()
    {
        try
        {
            var adapters = NetworkInterface.GetAllNetworkInterfaces()
                .Where(n =>
                    n.OperationalStatus == OperationalStatus.Up &&
                    n.NetworkInterfaceType != NetworkInterfaceType.Loopback &&
                    n.NetworkInterfaceType != NetworkInterfaceType.Tunnel)
                .OrderBy(n => n.Description)
                .ToList();

            foreach (var adapter in adapters)
            {
                string mac = adapter.GetPhysicalAddress().ToString();

                if (!string.IsNullOrWhiteSpace(mac))
                    return mac;
            }
        }
        catch
        {
        }

        return "UNKNOWN";
    }

    // ============================================================
    // SYSTEM UUID
    // ============================================================

    private static string GetSystemUuid()
    {
        return GetWmiValue(
            "Win32_ComputerSystemProduct",
            "UUID"
        );
    }

    // ============================================================
    // MOTHERBOARD SERIAL
    // ============================================================

    private static string GetMotherboardSerial()
    {
        return GetWmiValue(
            "Win32_BaseBoard",
            "SerialNumber"
        );
    }

    // ============================================================
    // BIOS SERIAL
    // ============================================================

    private static string GetBiosSerial()
    {
        return GetWmiValue(
            "Win32_BIOS",
            "SerialNumber"
        );
    }

    // ============================================================
    // DISK SERIAL
    // ============================================================

    private static string GetDiskSerial()
    {
        try
        {
            using var searcher = new ManagementObjectSearcher(
                "SELECT SerialNumber, DeviceID, MediaType " +
                "FROM Win32_DiskDrive"
            );

            var disks = searcher.Get()
                .Cast<ManagementObject>()
                .Where(x =>
                    x["SerialNumber"] != null &&
                    x["DeviceID"] != null)
                .OrderBy(x => x["DeviceID"]?.ToString())
                .ToList();

            foreach (var disk in disks)
            {
                string serial =
                    disk["SerialNumber"]?.ToString()?.Trim() ?? "";

                if (!string.IsNullOrWhiteSpace(serial))
                    return serial;
            }
        }
        catch
        {
        }

        return "UNKNOWN";
    }

    // ============================================================
    // GENERIC WMI READER
    // ============================================================

    private static string GetWmiValue(
        string className,
        string propertyName)
    {
        try
        {
            using var searcher = new ManagementObjectSearcher(
                $"SELECT {propertyName} FROM {className}"
            );

            foreach (ManagementObject obj in searcher.Get())
            {
                string value =
                    obj[propertyName]?.ToString()?.Trim() ?? "";

                if (!string.IsNullOrWhiteSpace(value))
                    return value;
            }
        }
        catch
        {
        }

        return "UNKNOWN";
    }

    // ============================================================
    // NORMALIZE VALUE
    // ============================================================

    private static string Normalize(string value)
    {
        if (string.IsNullOrWhiteSpace(value))
            return "UNKNOWN";

        return value
            .Trim()
            .ToUpperInvariant();
    }

    // ============================================================
    // MACHINE FINGERPRINT
    // ============================================================

    private static string GenerateFingerprint(MachineInfo info)
    {
        string combined =
            Normalize(info.MacAddress) +
            "|" +
            Normalize(info.SystemUuid) +
            "|" +
            Normalize(info.MotherboardSerial) +
            "|" +
            Normalize(info.BiosSerial) +
            "|" +
            Normalize(info.DiskSerial);

        using SHA256 sha = SHA256.Create();

        byte[] hash =
            sha.ComputeHash(
                Encoding.UTF8.GetBytes(combined)
            );

        return Convert.ToHexString(hash);
    }

    // ============================================================
    // CREATE REQUEST FOR LICENSE GENERATOR
    // ============================================================

    public static string CreateMachineRequest()
    {
        MachineInfo info = GetMachineInfo();

        return JsonSerializer.Serialize(
            info,
            new JsonSerializerOptions
            {
                WriteIndented = true
            }
        );
    }

    // ============================================================
    // SAVE REQUEST
    // ============================================================

    public static void SaveMachineRequest()
    {
        string json = CreateMachineRequest();

        string path =
            Path.Combine(
                AppContext.BaseDirectory,
                "machine_request.txt"
            );

        File.WriteAllText(path, json);

        Console.WriteLine();
        Console.WriteLine("======================================");
        Console.WriteLine(" MACHINE INFORMATION");
        Console.WriteLine("======================================");
        Console.WriteLine(json);
        Console.WriteLine();
        Console.WriteLine("Machine request saved to:");
        Console.WriteLine(path);
    }

    // ============================================================
    // LICENSE VERIFICATION
    // ============================================================

    public static bool VerifyLicense(string licenseFile)
    {
        try
        {
            string json =
                File.ReadAllText(licenseFile);

            LicenseData? license =
                JsonSerializer.Deserialize<LicenseData>(json);

            if (license == null)
                return false;

            // Check fingerprint
            MachineInfo current =
                GetMachineInfo();

            if (!string.Equals(
                current.Fingerprint,
                license.Fingerprint,
                StringComparison.OrdinalIgnoreCase))
            {
                Console.WriteLine(
                    "License is not registered to this computer."
                );

                return false;
            }

            // Check expiration
            if (DateTime.TryParse(
                license.Expiry,
                out DateTime expiry))
            {
                if (DateTime.UtcNow > expiry)
                {
                    Console.WriteLine(
                        "License has expired."
                    );

                    return false;
                }
            }

            // Verify cryptographic signature
            using RSA rsa = RSA.Create();

            rsa.ImportFromPem(PUBLIC_KEY);

            string data =
                license.LicenseId + "|" +
                license.Fingerprint + "|" +
                license.Product + "|" +
                license.Expiry;

            byte[] dataBytes =
                Encoding.UTF8.GetBytes(data);

            byte[] signature =
                Convert.FromBase64String(
                    license.Signature
                );

            bool valid =
                rsa.VerifyData(
                    dataBytes,
                    signature,
                    HashAlgorithmName.SHA256,
                    RSASignaturePadding.Pkcs1
                );

            if (!valid)
            {
                Console.WriteLine(
                    "Invalid license signature."
                );

                return false;
            }

            Console.WriteLine(
                "LICENSE VALID"
            );

            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine(
                "License verification failed: " +
                ex.Message
            );

            return false;
        }
    }
}

using System;
using System.IO;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;

public class MachineInfo
{
    public string MacAddress { get; set; } = "";
    public string SystemUuid { get; set; } = "";
    public string MotherboardSerial { get; set; } = "";
    public string BiosSerial { get; set; } = "";
    public string DiskSerial { get; set; } = "";
    public string Fingerprint { get; set; } = "";
}

public class LicenseData
{
    public string LicenseId { get; set; } = "";
    public string Fingerprint { get; set; } = "";
    public string Product { get; set; } = "";
    public string Expiry { get; set; } = "";
    public string Signature { get; set; } = "";
}

public static class LicenseGenerator
{
    private const string PRIVATE_KEY_FILE =
        "private_key.pem";

    private const string PUBLIC_KEY_FILE =
        "public_key.pem";

    // ============================================================
    // GENERATE RSA KEYS
    // ============================================================

    public static void GenerateKeys()
    {
        using RSA rsa = RSA.Create(3072);

        string privateKey =
            rsa.ExportRSAPrivateKeyPem();

        string publicKey =
            rsa.ExportSubjectPublicKeyInfoPem();

        File.WriteAllText(
            PRIVATE_KEY_FILE,
            privateKey
        );

        File.WriteAllText(
            PUBLIC_KEY_FILE,
            publicKey
        );

        Console.WriteLine();
        Console.WriteLine(
            "Keys generated successfully."
        );

        Console.WriteLine();
        Console.WriteLine(
            "PRIVATE KEY:"
        );

        Console.WriteLine(
            PRIVATE_KEY_FILE
        );

        Console.WriteLine();
        Console.WriteLine(
            "PUBLIC KEY:"
        );

        Console.WriteLine(
            publicKey
        );

        Console.WriteLine();
        Console.WriteLine(
            "Put ONLY the PUBLIC KEY into Client.cs."
        );

        Console.WriteLine(
            "NEVER distribute private_key.pem."
        );
    }

    // ============================================================
    // LOAD PRIVATE KEY
    // ============================================================

    private static RSA LoadPrivateKey()
    {
        if (!File.Exists(PRIVATE_KEY_FILE))
        {
            throw new Exception(
                "private_key.pem not found. " +
                "Run 'generate' first."
            );
        }

        string privateKey =
            File.ReadAllText(PRIVATE_KEY_FILE);

        RSA rsa = RSA.Create();

        rsa.ImportFromPem(privateKey);

        return rsa;
    }

    // ============================================================
    // CREATE LICENSE
    // ============================================================

    public static void GenerateLicense(
        string machineRequestFile,
        string licenseId,
        string product,
        DateTime expiry)
    {
        if (!File.Exists(machineRequestFile))
        {
            throw new FileNotFoundException(
                machineRequestFile
            );
        }

        string json =
            File.ReadAllText(machineRequestFile);

        MachineInfo? machine =
            JsonSerializer.Deserialize<MachineInfo>(
                json
            );

        if (machine == null)
        {
            throw new Exception(
                "Invalid machine request."
            );
        }

        if (string.IsNullOrWhiteSpace(
            machine.Fingerprint))
        {
            throw new Exception(
                "Machine fingerprint missing."
            );
        }

        LicenseData license =
            new LicenseData
            {
                LicenseId = licenseId,

                Fingerprint =
                    machine.Fingerprint,

                Product =
                    product,

                Expiry =
                    expiry.ToUniversalTime()
                        .ToString("O")
            };

        string data =
            license.LicenseId + "|" +
            license.Fingerprint + "|" +
            license.Product + "|" +
            license.Expiry;

        byte[] dataBytes =
            Encoding.UTF8.GetBytes(data);

        using RSA rsa = LoadPrivateKey();

        byte[] signature =
            rsa.SignData(
                dataBytes,
                HashAlgorithmName.SHA256,
                RSASignaturePadding.Pkcs1
            );

        license.Signature =
            Convert.ToBase64String(
                signature
            );

        string licenseJson =
            JsonSerializer.Serialize(
                license,
                new JsonSerializerOptions
                {
                    WriteIndented = true
                }
            );

        string output =
            "license_" +
            licenseId +
            ".json";

        File.WriteAllText(
            output,
            licenseJson
        );

        Console.WriteLine();
        Console.WriteLine(
            "======================================"
        );

        Console.WriteLine(
            "LICENSE GENERATED"
        );

        Console.WriteLine(
            "======================================"
        );

        Console.WriteLine();

        Console.WriteLine(
            licenseJson
        );

        Console.WriteLine();

        Console.WriteLine(
            "License file:"
        );

        Console.WriteLine(
            Path.GetFullPath(output)
        );
    }
}

        {
            Console.WriteLine(
                "Commands:"
            );

            Console.WriteLine();

            Console.WriteLine(
                "generate"
            );

            Console.WriteLine(
                "license <machine_request.txt> <license_id> <product> <expiry>"
            );

            Console.WriteLine();

            Console.WriteLine(
                "Example:"
            );

            Console.WriteLine(
                "license machine_request.txt ABC123 MySoftware 2027-12-31"
            );

            return;
        }

        try
        {
            if (args[0].Equals(
                "generate",
                StringComparison.OrdinalIgnoreCase))
            {
                LicenseGenerator.GenerateKeys();

                return;
            }

            if (args[0].Equals(
                "license",
                StringComparison.OrdinalIgnoreCase))
            {
                if (args.Length < 5)
                {
                    Console.WriteLine(
                        "Usage:"
                    );

                    Console.WriteLine(
                        "license <machine_request.txt> <license_id> <product> <expiry>"
                    );

                    return;
                }

                string machineFile = args[1];
                string licenseId = args[2];
                string product = args[3];

                DateTime expiry =
                    DateTime.Parse(args[4]);

                LicenseGenerator.GenerateLicense(
                    machineFile,
                    licenseId,
                    product,
                    expiry
                );

                return;
            }

            Console.WriteLine(
                "Unknown command."
            );
        }
        catch (Exception ex)
        {
            Console.WriteLine();
            Console.WriteLine(
                "ERROR: " + ex.Message
            );
        }
    }
